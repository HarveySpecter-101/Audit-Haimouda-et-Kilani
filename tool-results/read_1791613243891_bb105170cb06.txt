   158→const LONG_MONTHS = ['janvier', 'février', 'mars', 'avril', 'mai', 'juin', 'juillet', 'août', 'septembre', 'octobre', 'novembre', 'décembre']

   175→/** Phrase de synthèse adaptée au signe de l'écart de tentatives entre
   176→ *  la fenêtre courante et la fenêtre précédente (84 jours chacune). */
   177→function comparisonTakeaway(current: number, prev: number): string {
   178→  const diff = current - prev
   179→  if (diff > 0) {
   180→    return `Assiduité en hausse : +${diff} tentative${diff === 1 ? '' : 's'} par rapport aux 12 semaines précédentes.`
   181→  }
   182→  if (diff < 0) {
   183→    return `Assiduité en baisse : −${-diff} tentative${diff === -1 ? '' : 's'} par rapport aux 12 semaines précédentes.`
   184→  }
   185→  return current === 0
   186→    ? 'Aucune tentative sur aucune des deux périodes — la série démarre avec vous.'
   187→    : 'Assiduité stable : autant de tentatives que sur les 12 semaines précédentes.'
   188→}
   189→
   190→/** Couleur du delta (hausse / baisse / stable) — motif emerald/red établi
   191→ *  dans tout le simulateur, muted pour l'égalité. */
   192→function deltaTone(diff: number): string {
   193→  if (diff > 0) return 'text-emerald-700 dark:text-emerald-400'
   194→  if (diff < 0) return 'text-red-700 dark:text-red-400'
   195→  return 'text-muted-foreground'
   196→}
   197→
   198→/** F-9b — ligne de métrique comparée « libellé : X vs Y (delta) » :
   199→ *  delta +N (emerald), −N (red), = (muted), chiffres tabular-nums. */
   200→function ComparisonRow({ label, current, previous }: { label: string; current: number; previous: number }) {
   201→  const diff = current - previous
   202→  const delta = diff > 0 ? `+${diff}` : diff < 0 ? `−${-diff}` : '='
   203→  return (
   204→    <p className="flex flex-wrap items-baseline justify-between gap-x-3 gap-y-0.5">
   205→      <span className="text-muted-foreground">{label}</span>
   206→      <span className="tabular-nums">
   207→        <span className="font-medium text-foreground">{current}</span>
   208→        <span className="text-muted-foreground"> vs {previous}</span>
   209→        <span className={`ml-1.5 font-medium ${deltaTone(diff)}`}>({delta})</span>
   210→      </span>
   211→    </p>
   212→  )
   213→}

   219→/** F-9c — constructeur pur du CSV (testable unitairement) : BOM UTF-8 pour
   220→ *  Excel, séparateur point-virgule (Excel-FR), en-tête « Date;Tentatives »,
   221→ *  une ligne par jour « YYYY-MM-DD;count », ligne de total en fin — fins de
   222→ *  ligne CRLF (RFC 4180). Aucune dépendance, aucun accès DOM. */
   223→export function buildActivityCsv(days: ActivityDay[]): string {
   224→  const lines: string[] = ['Date;Tentatives']
   225→  let total = 0
   226→  for (const day of days) {
   227→    lines.push(`${day.date};${day.count}`)
   228→    total += day.count
   229→  }
   230→  lines.push(`Total;${total}`)
   231→  return `\uFEFF${lines.join('\r\n')}\r\n`
   232→}

   234→/** F-9c — téléchargement silencieux (Blob + ancre éphémère, aucune toast) :
   235→ *  l'ancre est attachée au document le temps du clic puis retirée, et
   236→ *  l'URL d'objet est révoquée immédiatement après le déclenchement. */
   237→function downloadActivityCsv(days: ActivityDay[]) {
   238→  const blob = new Blob([buildActivityCsv(days)], { type: 'text/csv;charset=utf-8' })
   239→  const url = URL.createObjectURL(blob)
   240→  const anchor = document.createElement('a')
   241→  anchor.href = url
   242→  anchor.download = `assiduite-${days.length}-jours.csv`
   243→  document.body.appendChild(anchor)
   244→  anchor.click()
   245→  anchor.remove()
   246→  URL.revokeObjectURL(url)
   247→}

   249→// Géométrie du viewBox (le SVG est responsive : w-full borné, h-auto)
   250→const CELL = 12
   251→const GAP = 3
   252→const PITCH = CELL + GAP
   253→const GX = 17 // gouttière des initiales de jours
   254→const GY = 15 // rangée des libellés de mois

   256→export function ActivityHeatmap({
   257→  days,
   258→  total,
   259→  activeDays,
   260→  streak,
   261→  todayCount,
   262→  today,
   263→  bestDay,
   264→  passed,
   265→  previous,
   266→}: ActivityHeatmapProps) {
   267→  const [selected, setSelected] = useState<string | null>(null)
   268→  // F-9b — panneau de comparaison ouvert/fermé (les métriques comparées
   269→  // sont intégralement DÉRIVÉES des props : zéro effet, zéro état dérivé
   270→  // synchronisé — seul l'ouverture est un état local).
   271→  const [compareOpen, setCompareOpen] = useState(false)
   272→  // F-8c — filtre par mode du panneau du jour, clé par date : le mode
   273→  // effectif est DÉRIVÉ (dayFilter.date === selected ? mode : 'all'),
   274→  // changer de journée retombe naturellement sur « Tous » sans le
   275→  // moindre setState dans un effet (règle react-hooks/set-state-in-effect).
   276→  const [dayFilter, setDayFilter] = useState<{ date: string; mode: DayFilterMode }>({ date: '', mode: 'all' })
   277→  // F-8d — horodatage du dernier « g » nu capté dans le composant :
   278→  // miroir local de l'armement des raccourcis globaux (g puis lettre,
   279→  // fenêtre ~900 ms) afin de céder « d » à la séquence de navigation
   280→  // quand elle vient d'être armée (cf. handleRootKeyDown).
   281→  const gArmedAtRef = useRef(0)
   282→  const navigate = useSimulator((s) => s.navigate)
   283→
   284→  // Détail des tentatives de la journée sélectionnée (drill-down)
   285→  const dayQuery = useQuery<{ attempts: DayAttempt[] }>({
   286→    queryKey: ['attempts-day', selected],
   287→    enabled: !!selected,
   288→    queryFn: async () => {
   289→      const res = await fetch(`/api/attempts?date=${selected}&limit=100`)
   290→      if (!res.ok) throw new Error('Erreur de lecture des tentatives du jour.')
   291→      return res.json()
   292→    },
   293→    refetchOnWindowFocus: false,
   294→    staleTime: 30_000,
   295→  })
   296→
   297→  if (!days.length) return null
   298→
   299→  const todayN = todayCount ?? today ?? 0
   300→  const best = bestDay ?? days.reduce((m, d) => Math.max(m, d.count), 0)
   301→  // « aujourd'hui » = dernière entrée de days (convention du composant)
   302→  const todayDate = days[days.length - 1].date
   303→  const todayPanelOpen = selected === todayDate
   304→
   305→  // Analyse locale des dates — jamais new Date(iso) : pivot UTC selon le fuseau
   306→  const cells = days.map((d) => {
   307→    const [ys, ms, ds] = d.date.split('-')
   308→    return { date: d.date, count: d.count, at: new Date(Number(ys), Number(ms) - 1, Number(ds)) }
   309→  })
   310→  // Alignement lundi : décalage de la première colonne (cellules vides avant la fenêtre)
   311→  const offset = (cells[0].at.getDay() + 6) % 7
   312→  const cols = Math.ceil((offset + cells.length) / 7)
   313→
   314→  const W = GX + cols * PITCH - GAP + 3
   315→  const H = GY + 7 * PITCH - GAP + 3
   316→
   317→  // Libellé de mois au-dessus d'une colonne quand la semaine change de mois
   318→  // (mois du DERNIER jour réel de la colonne — couvre un mois qui débute en
   319→  // cours de semaine) ; garde-fou anti-chevauchement de libellés adjacents
   320→  const monthLabels: { col: number; label: string }[] = []
   321→  let lastLabelRight = -Infinity
   322→  for (let c = 0; c < cols; c++) {
   323→    const lastSlot = Math.min((c + 1) * 7 - 1, offset + cells.length - 1)
   324→    const month = cells[lastSlot - offset].at.getMonth()
   325→    const prevLastSlot = c === 0 ? -1 : Math.min(c * 7 - 1, offset + cells.length - 1)
   326→    const prevMonth = prevLastSlot < offset ? -1 : cells[prevLastSlot - offset].at.getMonth()
   327→    if (month === prevMonth) continue
   328→    const label = MONTH_LABELS[month]
   329→    const center = GX + c * PITCH + CELL / 2
   330→    const halfW = (label.length * 4.6) / 2
   331→    if (center - halfW < lastLabelRight + 1.5) continue
   332→    monthLabels.push({ col: c, label })
   333→    lastLabelRight = center + halfW
   334→  }
   335→
   336→  const summary =
   337→    `Assiduité sur ${cells.length} jours : ${total} tentatives au total, ${activeDays} jours actifs, ` +
   338→    `${best} tentatives au maximum en une journée, série en cours de ${streak} jour${streak === 1 ? '' : 's'}, ` +
   339→    `${todayN} tentative${todayN === 1 ? '' : 's'} aujourd'hui. Cliquez sur une journée active pour voir le détail des tentatives.`
   340→
   341→  const toggleDay = (date: string) => {
   342→    setSelected((cur) => (cur === date ? null : date))
   343→  }
   344→
   345→  // ---- F-8c : vue filtrée du panneau du jour (état dérivé, zéro effet) ----
   346→  const filterMode: DayFilterMode = dayFilter.date === selected ? dayFilter.mode : 'all'
   347→  const dayAttempts = dayQuery.data?.attempts ?? []
   348→  const filteredAttempts =
   349→    filterMode === 'all' ? dayAttempts : dayAttempts.filter((a) => a.mode === filterMode)
   350→  const modeCounts: Record<Exclude<DayFilterMode, 'all'>, number> = { guided: 0, mission: 0, exam: 0 }
   351→  for (const a of dayAttempts) {
   352→    if (a.mode === 'guided' || a.mode === 'mission' || a.mode === 'exam') modeCounts[a.mode]++
   353→  }
   354→
   355→  // ---- F-8d : raccourci local « D » (aujourd'hui) ----
   356→  // POURQUOI un « d » nu est sûr : les raccourcis GLOBAUX
   357→  // (keyboard-shortcuts.tsx) sont soit des SÉQUENCES « g » puis lettre
   358→  // (≤ 900 ms), soit des touches isolées qui ne sont PAS « d » ([ ] / c ?
   359→  // F1 — aucun Alt+lettre n'existe dans l'app). Un « d » tapé seul ne peut
   360→  // donc déclencher aucune action globale. Seul cas d'intersection : « d »
   361→  // comme SECONDE touche d'une séquence « g d » armée — on reflète localement
   362→  // l'armement (gArmedAtRef) et on laisse alors filer la frappe vers le
   363→  // gestionnaire global (navigation Tableau de bord) au lieu d'ouvrir le panneau.
   364→  const handleRootKeyDown = (e: React.KeyboardEvent<HTMLDivElement>) => {
   365→    // Frappe dans un champ de saisie : ce n'est pas un raccourci.
   366→    const t = e.target as HTMLElement | null
   367→    if (t && (t.tagName === 'INPUT' || t.tagName === 'TEXTAREA' || t.tagName === 'SELECT' || t.isContentEditable)) {
   368→      return
   369→    }
   370→    // Modificateurs ou Maj : ce n'est plus un « d » nu (Ctrl/Cmd+D = favori,
   371→    // Alt+D… rien à faire ici — on garde la frappe pour le navigateur).
   372→    if (e.altKey || e.ctrlKey || e.metaKey || e.shiftKey) return
   373→    if (e.key === 'g' || e.key === 'G') {
   374→      gArmedAtRef.current = Date.now()
   375→      return
   376→    }
   377→    if (e.key !== 'd' && e.key !== 'D') return
   378→    // Séquence « g d » en cours d'armement : la frappe appartient à la
   379→    // navigation globale — on ne consomme pas l'événement.
   380→    if (Date.now() - gArmedAtRef.current < 900) return
   381→    // Aucune tentative aujourd'hui : rien à ouvrir (le chip reste non cliquable).
   382→    if (todayN === 0) return
   383→    e.preventDefault()
   384→    toggleDay(todayDate)
   385→  }
   386→
   387→  return (
   388→    <div
   389→      className="activity-heatmap space-y-3 rounded-lg focus-visible:outline-2 focus-visible:outline-offset-2 focus-visible:outline-primary/60"
   390→      tabIndex={0}
   391→      role="group"
   392→      aria-label="Assiduité — tentatives des 12 dernières semaines"
   393→      onKeyDown={handleRootKeyDown}
   394→    >
   395→      {/* Chips de statistiques */}
   396→      <div className="flex flex-wrap items-center gap-2 text-xs text-muted-foreground">
   397→        <span className="flex items-center gap-1.5 rounded-full border bg-muted/40 px-2.5 py-1">
   398→          <Flame className="size-3.5 text-amber-600" aria-hidden />
   399→          série : <span className="font-medium tabular-nums text-foreground">{streak}</span> jour{streak === 1 ? '' : 's'}
   400→        </span>
   401→        {/* F-8d — chip « aujourd'hui » : vrai bouton (toggle du panneau du
   402→            jour) dès qu'il y a une tentative ; sinon simple span informatif. */}
   403→        {todayN > 0 ? (
   404→          <button
   405→            type="button"
   406→            onClick={() => toggleDay(todayDate)}
   407→            aria-label={
   408→              todayPanelOpen
   409→                ? `Fermer le détail des ${todayN} tentative${todayN === 1 ? '' : 's'} d’aujourd’hui`
   410→                : `Voir les ${todayN} tentative${todayN === 1 ? '' : 's'} d’aujourd’hui`
   411→            }
   412→            className="flex items-center gap-1.5 rounded-full border bg-muted/40 px-2.5 py-1 transition-colors hover:border-primary/40 hover:bg-primary/5 focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring/60"
