    80→    80→export function ExerciseRunner({ exercise }: { exercise: SafeExerciseData }) {
    81→    81→  const { toast } = useToast()
    82→    82→  const queryClient = useQueryClient()
    83→    83→  const mode = useSimulator((s) => s.mode)
    84→    84→  const [answer, setAnswer] = useState<AttemptInputDTO>({})
    85→    85→  const [feedback, setFeedback] = useState<AttemptFeedback | null>(null)
    86→    86→  const [hintLevel, setHintLevel] = useState(0)
    87→    87→  const [showExample, setShowExample] = useState(false)
    88→    88→  const [attempted, setAttempted] = useState(false)
    89→    89→
    90→    90→  const mutation = useMutation({
    91→    91→    mutationFn: async () => {
    92→    92→      const res = await fetch('/api/attempts', {
    93→    93→        method: 'POST',
    94→    94→        headers: { 'Content-Type': 'application/json' },
    95→    95→        body: JSON.stringify({ exerciseKey: exercise.key, answer, usedHints: hintLevel, mode }),
    96→    96→      })
    97→    97→      if (!res.ok) throw new Error('Erreur de correction')
    98→    98→      return res.json() as Promise<{ feedback: AttemptFeedback }>
    99→    99→    },
   100→   100→    onSuccess: (data) => {
   101→   101→      setFeedback(data.feedback)
   102→   102→      setAttempted(true)
   103→   103→      queryClient.invalidateQueries({ queryKey: ['dashboard'] })
   104→   104→      queryClient.invalidateQueries({ queryKey: ['attempts'] })
   105→   105→      queryClient.invalidateQueries({ queryKey: ['progress'] })
   106→   106→      if (data.feedback.passed) {
   107→   107→        toast({ title: 'Exercice réussi', description: `Score : ${Math.round(data.feedback.score * 100)} %` })
   108→   108→      } else {
   109→   109→        toast({
   110→   110→          title: 'Tentative enregistrée',
   111→   111→          description: 'Lisez le feedback et retentez : les tentatives multiples sont autorisées.',
   112→   112→          variant: 'destructive',
   113→   113→        })
   114→   114→      }
   115→   115→    },
   116→   116→    onError: () => {
   117→   117→      toast({ title: 'Erreur', description: 'La correction a échoué. Réessayez.', variant: 'destructive' })
   118→   118→    },
   119→   119→  })
   120→   120→
   121→   121→  const difficultyLabel = useMemo(
   122→   122→    () =>
   123→   123→      ({ SHORT: 'Cas court', INTERMEDIATE: 'Intermédiaire', SYNTHESIS: 'Cas de synthèse' })[exercise.difficulty] ??
   124→   124→      exercise.difficulty,
   125→   125→    [exercise.difficulty],
   126→   126→  )
   127→   127→
   128→   128→  const canSubmit = useMemo(() => {
   129→   129→    switch (exercise.kind) {
   130→   130→      case 'qcm':
   131→   131→        return (answer.qcm?.choiceIds?.length ?? 0) > 0
   132→   132→      case 'entry':
   133→   133→        return (answer.entry?.lines?.length ?? 0) >= 2
   134→   134→      case 'calculation':
   135→   135→        return Object.keys(answer.calculation ?? {}).length >= (exercise.fields?.length ?? 1)
   136→   136→      case 'matching':
   137→   137→        return Object.keys(answer.matching ?? {}).length >= (exercise.left?.length ?? 1)
   138→   138→      case 'detection':
   139→   139→        return true
   140→   140→      case 'ordering':
   141→   141→        return (answer.ordering?.ids?.length ?? 0) === (exercise.items?.length ?? 1)
   142→   142→      case 'essay':
   143→   143→        return (answer.essay?.text?.trim().length ?? 0) > 20
   144→   144→      case 'case':
   145→   145→        return true
   146→   146→      default:
   147→   147→        return false
   148→   148→    }
   149→   149→  }, [answer, exercise])
   150→   150→
   151→   151→  return (
   152→   152→    <Card className="gap-4">
   153→   153→      <CardHeader className="pb-3">
   154→   154→        <div className="flex flex-wrap items-center gap-2">
   155→   155→          <CardTitle className="text-base">{exercise.title}</CardTitle>
   156→   156→          <Badge variant="outline">{difficultyLabel}</Badge>
   157→   157→          <Badge variant="secondary">{exercise.points} pts</Badge>
   158→   158→          {exercise.critical ? <Badge className="gap-1"><AlertTriangle className="size-3" /> Critique</Badge> : null}
   159→   159→        </div>
   160→   160→        {exercise.context ? <p className="text-sm text-muted-foreground">{exercise.context}</p> : null}
   161→   161→        {exercise.documents?.length ? (
   162→   162→          <p className="text-xs text-muted-foreground">
   163→   163→            Pièces à consulter (centre documentaire) : {exercise.documents.join(', ')}
   164→   164→          </p>
   165→   165→        ) : null}
   166→   166→      </CardHeader>
   167→   167→      <CardContent className="space-y-4">
   168→   168→        <p className="text-sm leading-relaxed">{exercise.prompt}</p>
   169→   169→
   170→   170→        <ExerciseBody exercise={exercise} answer={answer} setAnswer={setAnswer} disabled={attempted && (feedback?.passed ?? false)} />
   171→   171→
   172→   172→        <Separator />
   173→   173→
   174→   174→        <div className="flex flex-wrap items-center gap-2">
   175→   175→          <Button
   176→   176→            onClick={() => mutation.mutate()}
   177→   177→            disabled={!canSubmit || mutation.isPending || (attempted && (feedback?.passed ?? false))}
   178→   178→            className="gap-1.5"
   179→   179→          >
   180→   180→            <Send className="size-4" />
   181→   181→            {mutation.isPending ? 'Correction…' : 'Soumettre ma réponse'}
   182→   182→          </Button>
   183→   183→          {exercise.hints.length > 0 ? (
   184→   184→            <Button
   185→   185→              variant="outline"
   186→   186→              size="sm"
   187→   187→              className="gap-1.5"
   188→   188→              onClick={() => setHintLevel(Math.min(exercise.hints.length, hintLevel + 1))}
   189→   189→              disabled={hintLevel >= exercise.hints.length}
   190→   190→            >
   191→   191→              <Lightbulb className="size-3.5" /> Indice ({hintLevel}/{exercise.hints.length})
   192→   192→            </Button>
   193→   193→          ) : null}
   194→   194→          {exercise.hasExample ? (
   195→   195→            <Button
   196→   196→              variant="ghost"
   197→   197→              size="sm"
   198→   198→              className="gap-1.5"
   199→   199→              onClick={() => setShowExample(!showExample)}
   200→   200→              disabled={!attempted}
   201→   201→              title={attempted ? 'Exemple corrigé (après une tentative)' : 'Disponible après une tentative'}
   202→   202→            >
   203→   203→              <Eye className="size-3.5" /> Exemple corrigé
   204→   204→            </Button>
   205→   205→          ) : null}
   206→   206→          {attempted && !feedback?.passed ? (
   207→   207→            <Button
   208→   208→              variant="ghost"
   209→   209→              size="sm"
   210→   210→              className="gap-1.5"
   211→   211→              onClick={() => {
   212→   212→                setFeedback(null)
   213→   213→                setAttempted(false)
   214→   214→                setAnswer({})
   215→   215→              }}
   216→   216→            >
   217→   217→              <RefreshCw className="size-3.5" /> Recommencer
   218→   218→            </Button>
   219→   219→          ) : null}
   220→   220→        </div>
   221→   221→
   222→   222→        {hintLevel > 0 && (
   223→   223→          <div className="space-y-1 rounded-md border border-amber-200 bg-amber-50 p-3 dark:border-amber-900 dark:bg-amber-950">
   224→   224→            {exercise.hints.slice(0, hintLevel).map((h, i) => (
   225→   225→              <p key={i} className="text-sm text-amber-900 dark:text-amber-200">
   226→   226→                <span className="font-semibold">Indice {i + 1} : </span>
   227→   227→                {h}
   228→   228→              </p>
   229→   229→            ))}
   230→   230→          </div>
   231→   231→        )}
   232→   232→
   233→   233→        {showExample && exercise.exampleSolution ? (
   234→   234→          <div className="rounded-md border bg-muted p-3">
   235→   235→            <p className="mb-1 text-xs font-semibold uppercase tracking-wide text-muted-foreground">Exemple corrigé</p>
   236→   236→            <p className="font-mono text-xs">{exercise.exampleSolution}</p>
   237→   237→          </div>
   238→   238→        ) : null}
   239→   239→
   240→   240→        {feedback ? <FeedbackPanel feedback={feedback} /> : null}
   241→   241→      </CardContent>
   242→   242→    </Card>
   243→   243→  )
   244→   244→}

   248→   248→function ExerciseBody({
   249→   249→  exercise,
   250→   250→  answer,
   251→   251→  setAnswer,
   252→   252→  disabled,
   253→   253→}: {
   254→   254→  exercise: SafeExerciseData
   255→   255→  answer: AttemptInputDTO
   256→   256→  setAnswer: (a: AttemptInputDTO) => void
   257→   257→  disabled: boolean
   258→   258→}) {
   259→   259→  switch (exercise.kind) {
   260→   260→    case 'qcm':
   261→   261→      return <QcmBody exercise={exercise} answer={answer} setAnswer={setAnswer} disabled={disabled} />
   262→   262→    case 'entry':
   263→   263→      return <EntryBody answer={answer} setAnswer={setAnswer} disabled={disabled} />
   264→   264→    case 'calculation':
   265→   265→      return <CalculationBody exercise={exercise} answer={answer} setAnswer={setAnswer} disabled={disabled} />
   266→   266→    case 'matching':
   267→   267→      return <MatchingBody exercise={exercise} answer={answer} setAnswer={setAnswer} disabled={disabled} />
   268→   268→    case 'detection':
   269→   269→      return <DetectionBody exercise={exercise} answer={answer} setAnswer={setAnswer} disabled={disabled} />
   270→   270→    case 'ordering':
   271→   271→      return <OrderingBody exercise={exercise} answer={answer} setAnswer={setAnswer} disabled={disabled} />
   272→   272→    case 'essay':
   273→   273→      return <EssayBody exercise={exercise} answer={answer} setAnswer={setAnswer} disabled={disabled} />
   274→   274→    case 'case':
   275→   275→      return <CaseBody exercise={exercise} answer={answer} setAnswer={setAnswer} disabled={disabled} />
   276→   276→    default:
   277→   277→      return null
   278→   278→  }
   279→   279→}

   281→   281→function QcmBody({
   282→   282→  exercise,
   283→   283→  answer,
   284→   284→  setAnswer,
   285→   285→  disabled,
   286→   286→}: {
   287→   287→  exercise: SafeExerciseData
   288→   288→  answer: AttemptInputDTO
   289→   289→  setAnswer: (a: AttemptInputDTO) => void
   290→   290→  disabled: boolean
   291→   291→}) {
   292→   292→  const selected = answer.qcm?.choiceIds ?? []
   293→   293→  const toggle = (id: string) => {
   294→   294→    if (disabled) return
   295→   295→    const next = exercise.multi
   296→   296→      ? selected.includes(id)
   297→   297→        ? selected.filter((s) => s !== id)
   298→   298→        : [...selected, id]
   299→   299→      : [id]
   300→   300→    setAnswer({ ...answer, qcm: { choiceIds: next } })
   301→   301→  }
   302→   302→  if (exercise.multi) {
   303→   303→    return (
   304→   304→      <div className="space-y-2">
   305→   305→        {exercise.choices?.map((c) => (
   306→   306→          <label key={c.id} className="flex cursor-pointer items-start gap-2.5 rounded-md border p-3 text-sm hover:bg-accent/40">
   307→   307→            <Checkbox checked={selected.includes(c.id)} onCheckedChange={() => toggle(c.id)} disabled={disabled} />
   308→   308→            <span>{c.text}</span>
   309→   309→          </label>
   310→   310→        ))}
   311→   311→        <p className="text-xs text-muted-foreground">Plusieurs réponses possibles.</p>
   312→   312→      </div>
   313→   313→    )
   314→   314→  }
   315→   315→  return (
   316→   316→    <RadioGroup value={selected[0] ?? ''} onValueChange={(v) => !disabled && setAnswer({ ...answer, qcm: { choiceIds: [v] } })} className="gap-2">
   317→   317→      {exercise.choices?.map((c) => (
   318→   318→        <label key={c.id} className="flex cursor-pointer items-start gap-2.5 rounded-md border p-3 text-sm hover:bg-accent/40">
   319→   319→          <RadioGroupItem value={c.id} disabled={disabled} />
   320→   320→          <span>{c.text}</span>
   321→   321→        </label>
   322→   322→      ))}
   323→   323→    </RadioGroup>
   324→   324→  )

   327→   327→interface DraftLine {

   330→   330→  debit: string

   334→   334→function EntryBody({
   335→   335→  answer,
   336→   336→  setAnswer,
   337→   337→  disabled,
   338→   338→}: {
   339→   339→  answer: AttemptInputDTO
   340→   340→  setAnswer: (a: AttemptInputDTO) => void
   341→   341→  disabled: boolean
   342→   342→}) {
   343→   343→  const [journal, setJournal] = useState(answer.entry?.journal ?? 'ACH')
   344→   344→  const [lines, setLines] = useState<DraftLine[]>(
   345→   345→    answer.entry?.lines?.map((l) => ({
   346→   346→      accountCode: l.accountCode,
   347→   347→      label: l.label ?? '',
   348→   348→      debit: l.debit ? String(l.debit) : '',
   349→   349→      credit: l.credit ? String(l.credit) : '',
   350→   350→    })) ?? [
   351→   351→      { accountCode: '', label: '', debit: '', credit: '' },
   352→   352→      { accountCode: '', label: '', debit: '', credit: '' },
   353→   353→      { accountCode: '', label: '', debit: '', credit: '' },
   354→   354→    ],
   355→   355→  )
   356→   356→
   357→   357→  const totalDebit = round2(lines.reduce((a, l) => a + (Number(l.debit.replace(',', '.')) || 0), 0))
   358→   358→  const totalCredit = round2(lines.reduce((a, l) => a + (Number(l.credit.replace(',', '.')) || 0), 0))
   359→   359→  const balanced = Math.abs(totalDebit - totalCredit) < 0.005 && totalDebit > 0
   360→   360→
   361→   361→  const pushToAnswer = (nextLines: DraftLine[], nextJournal: string) => {
   362→   362→    setAnswer({
   363→   363→      ...answer,
   364→   364→      entry: {
   365→   365→        journal: nextJournal,
   366→   366→        lines: nextLines
   367→   367→          .filter((l) => l.accountCode && (Number(l.debit.replace(',', '.')) > 0 || Number(l.credit.replace(',', '.')) > 0))
   368→   368→          .map((l) => ({
   369→   369→            accountCode: l.accountCode,
   370→   370→            label: l.label,
   371→   371→            debit: Number(l.debit.replace(',', '.')) || 0,
   372→   372→            credit: Number(l.credit.replace(',', '.')) || 0,
   373→   373→          })),
   374→   374→      },
   375→   375→    })
   376→   376→  }
   377→   377→
   378→   378→  const updateLine = (i: number, patch: Partial<DraftLine>) => {
   379→   379→    const next = lines.map((l, idx) => (idx === i ? { ...l, ...patch } : l))
   380→   380→    setLines(next)
   381→   381→    pushToAnswer(next, journal)
   382→   382→  }
   383→   383→
   384→   384→  return (
   385→   385→    <div className="space-y-3">
   386→   386→      <div className="flex flex-wrap items-center gap-3">
   387→   387→        <div className="w-40">
   388→   388→          <Label className="text-xs">Journal</Label>
   389→   389→          <Select value={journal} onValueChange={(v) => { setJournal(v); pushToAnswer(lines, v) }} disabled={disabled}>
   390→   390→            <SelectTrigger><SelectValue /></SelectTrigger>
   391→   391→            <SelectContent>
   392→   392→              {['ACH', 'VTE', 'BQE', 'CAI', 'OD', 'PAIE'].map((j) => (
   393→   393→                <SelectItem key={j} value={j}>{j}</SelectItem>
   394→   394→              ))}
   395→   395→            </SelectContent>
   396→   396→          </Select>
   397→   397→        </div>
   398→   398→        <div className="ml-auto flex items-center gap-3 text-sm">
   399→   399→          <span className="tabular-nums">
   400→   400→            Σ Débit <strong className={balanced ? 'text-emerald-700' : 'text-red-700'}>{fmtEUR(totalDebit)}</strong>
   401→   401→          </span>
   402→   402→          <span className="tabular-nums">
   403→   403→            Σ Crédit <strong className={balanced ? 'text-emerald-700' : 'text-red-700'}>{fmtEUR(totalCredit)}</strong>
   404→   404→          </span>
   405→   405→          {balanced ? (
   406→   406→            <Badge className="gap-1 bg-emerald-600"><CheckCircle2 className="size-3" /> Équilibrée</Badge>
   407→   407→          ) : (
   408→   408→            <Badge variant="destructive" className="gap-1"><XCircle className="size-3" /> Déséquilibrée</Badge>
   409→   409→          )}
   410→   410→        </div>
   411→   411→      </div>
   412→   412→      <div className="overflow-x-auto">
   413→   413→        <table className="w-full min-w-[640px] text-sm">
   414→   414→          <thead>
   415→   415→            <tr className="border-b text-left text-xs uppercase tracking-wide text-muted-foreground">
   416→   416→              <th className="pb-2 pr-2">Compte</th>
   417→   417→              <th className="pb-2 pr-2">Libellé</th>
   418→   418→              <th className="pb-2 pr-2 w-28 text-right">Débit</th>
   419→   419→              <th className="pb-2 pr-2 w-28 text-right">Crédit</th>
   420→   420→            </tr>
   421→   421→          </thead>
   422→   422→          <tbody>
   423→   423→            {lines.map((l, i) => (
   424→   424→              <tr key={i} className="border-b border-dashed">
   425→   425→                <td className="py-1.5 pr-2">
   426→   426→                  <Select value={l.accountCode} onValueChange={(v) => updateLine(i, { accountCode: v })} disabled={disabled}>
   427→   427→                    <SelectTrigger className="h-8 text-xs"><SelectValue placeholder="Choisir…" /></SelectTrigger>
   428→   428→                    <SelectContent>
   429→   429→                      <ScrollArea className="h-64">
   430→   430→                        {COMMON_ENTRY_ACCOUNTS.map((code) => (
   431→   431→                          <SelectItem key={code} value={code} className="text-xs">
   432→   432→                            {code} — {accountLabel(code).slice(0, 44)}
   433→   433→                          </SelectItem>
   434→   434→                        ))}
   435→   435→                      </ScrollArea>
   436→   436→                    </SelectContent>
   437→   437→                  </Select>
   438→   438→                </td>
   439→   439→                <td className="py-1.5 pr-2">
   440→   440→                  <Input

   596→   596→function OrderingBody({
   597→   597→  exercise,
   598→   598→  answer,
   599→   599→  setAnswer,
   600→   600→  disabled,
   601→   601→}: {
   602→   602→  exercise: SafeExerciseData
   603→   603→  answer: AttemptInputDTO
   604→   604→  setAnswer: (a: AttemptInputDTO) => void
   605→   605→  disabled: boolean
   606→   606→}) {
   607→   607→  const selected = answer.ordering?.ids ?? []
   608→   608→  const items = exercise.items ?? []
   609→   609→  const move = (id: string, dir: -1 | 1) => {
   610→   610→    if (disabled) return
   611→   611→    const idx = selected.indexOf(id)
   612→   612→    const next = [...selected]
   613→   613→    const target = idx + dir
   614→   614→    if (idx >= 0 && target >= 0 && target < next.length) {
   615→   615→      ;[next[idx], next[target]] = [next[target], next[idx]]
   616→   616→      setAnswer({ ...answer, ordering: { ids: next } })
   617→   617→    }
   618→   618→  }
   619→   619→  const add = (id: string) => {
   620→   620→    if (disabled || selected.includes(id)) return
   621→   621→    setAnswer({ ...answer, ordering: { ids: [...selected, id] } })
   622→   622→  }
   623→   623→  return (
   624→   624→    <div className="grid gap-4 sm:grid-cols-2">
   625→   625→      <div className="space-y-1.5">
   626→   626→        <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Disponibles</p>
   627→   627→        {items
   628→   628→          .filter((it) => !selected.includes(it.id))
   629→   629→          .map((it) => (
   630→   630→            <button
   631→   631→              key={it.id}
   632→   632→              className="flex w-full items-center justify-between rounded-md border p-2 text-left text-sm hover:bg-accent/40"
   633→   633→              onClick={() => add(it.id)}
   634→   634→            >
   635→   635→              <span>{it.label}</span>
   636→   636→              <span className="text-xs text-muted-foreground">+ ajouter</span>
   637→   637→            </button>
   638→   638→          ))}
   639→   639→      </div>
   640→   640→      <div className="space-y-1.5">
   641→   641→        <p className="text-xs font-semibold uppercase tracking-wide text-muted-foreground">Ordre proposé</p>
   642→   642→        {selected.map((id, i) => {
   643→   643→          const it = items.find((x) => x.id === id)
   644→   644→          return (
   645→   645→            <div key={id} className="flex items-center gap-2 rounded-md border p-2 text-sm">
   646→   646→              <span className="tabular-nums w-5 text-center text-xs text-muted-foreground">{i + 1}</span>
   647→   647→              <span className="flex-1">{it?.label}</span>
   648→   648→              <span className="flex gap-0.5">
   649→   649→                <Button variant="ghost" size="icon" className="size-6" onClick={() => move(id, -1)} aria-label="Monter">↑</Button>
   650→   650→                <Button variant="ghost" size="icon" className="size-6" onClick={() => move(id, 1)} aria-label="Descendre">↓</Button>
   651→   651→              </span>
   652→   652→            </div>
   653→   653→          )
   654→   654→        })}
   655→   655→      </div>
   656→   656→    </div>
   657→   657→  )
   658→   658→}
