    58→    58→  { value: '0.9', label: '90 %', z: 'z = 1,645' },

    62→    62→interface SalesEntry {
    63→    63→  id: string
    64→    64→  pieceRef: string
    65→    65→  date: string
    66→    66→  label: string
    67→    67→  lines: { accountCode: string; debit: number; credit: number }[]
    68→    68→}
    69→    69→
    70→    70→function itemBook(e: SalesEntry): number {
    71→    71→  const deb411 = e.lines.filter((l) => l.accountCode.startsWith('411')).reduce((s, l) => s + l.debit, 0)
    72→    72→  if (deb411 > 0) return deb411
    73→    73→  return e.lines.filter((l) => l.accountCode.startsWith('70')).reduce((s, l) => s + l.credit, 0)
    74→    74→}
    75→    75→
    76→    76→const toState = (it: PopulationItem & { date?: string; label?: string }): SampleItemState => ({
    77→    77→  ref: it.ref,
    78→    78→  book: it.book,
    79→    79→  status: 'PENDING',
    80→    80→  error: 0,
    81→    81→})
    82→    82→
    83→    83→export function SamplingTool() {
    84→    84→  const { ws, save } = useWorksheet()
    85→    85→
    86→    86→  // ---- Population réelle : journal des ventes ----
    87→    87→  const sales = useQuery<{ entries: SalesEntry[]; total: number }>({
    88→    88→    queryKey: ['sales-population-full'],
    89→    89→    queryFn: async () => {
    90→    90→      const res = await fetch('/api/entries?journal=VTE&limit=300')
    91→    91→      if (!res.ok) throw new Error('Erreur')
    92→    92→      return res.json()
    93→    93→    },
    94→    94→  })
    95→    95→
    96→    96→  const population = useMemo<PopulationItem[]>(() => {
    97→    97→    const all = (sales.data?.entries ?? []).map((e) => ({
    98→    98→      ref: e.pieceRef,
    99→    99→      book: itemBook(e),
   100→   100→      label: e.label,
   101→   101→      date: e.date,
   102→   102→    }))
   103→   103→    // La population monétaire exclut avoirs et opérations diverses (valeur ≤ 0)
   104→   104→    return all.filter((it) => it.book > 0)
   105→   105→  }, [sales.data])
   106→   106→
   107→   107→  // ---- Seuils de la mission (feuille « Seuils » si sauvegardée, sinon défaut NEP 320) ----
   108→   108→  const materiality = ws.data?.worksheet.materiality
   109→   109→  const thresholds = useMemo(
   110→   110→    () =>
   111→   111→      materiality
   112→   112→        ? { signification: materiality.seuil, planification: materiality.planification, remontee: materiality.remontee, saved: true }
   113→   113→        : {
   114→   114→            signification: 44071.13,
   115→   115→            planification: 28646.23,
   116→   116→            remontee: 2203.56,
   117→   117→            saved: false,
   118→   118→          },
   119→   119→    [materiality],
   120→   120→  )
   121→   121→
   122→   122→  // ---- Paramètres de sondage ----
   123→   123→  const savedSampling: SamplingWorksheet | undefined = ws.data?.worksheet.sampling
   124→   124→  const [method, setMethod] = useState<SamplingMethod>(savedSampling?.method ?? 'mus')
   125→   125→  const [confidence, setConfidence] = useState<ConfidenceLevel>(savedSampling?.confidence ?? 0.95)
   126→   126→  const [seedStr, setSeedStr] = useState(String(savedSampling?.seed ?? 2024))
   127→   127→  const [nStr, setNStr] = useState(String(savedSampling?.taille ?? 25))
   128→   128→  const [hvStr, setHvStr] = useState('20000')
   129→   129→  // Overrides locaux (null = reprise de la feuille sauvegardée, dérivée
   130→   130→  // des données — pas d'effet de synchronisation nécessaire)
   131→   131→  const [sampleOverride, setSampleOverride] = useState<TestResult[] | null>(null)
   132→   132→  const [hvOverride, setHvOverride] = useState<TestResult[] | null>(null)
   133→   133→
   134→   134→  // Échantillon restauré depuis la feuille (format complet, ou anciennes
   135→   135→  // références seules converties en items « à tester » dès que la
   136→   136→  // population est chargée)
   137→   137→  const savedSample = useMemo<TestResult[]>(() => {
   138→   138→    if (!savedSampling) return []
   139→   139→    if (savedSampling.items?.length) return savedSampling.items
   140→   140→    if (!sales.isSuccess) return []
   141→   141→    const byRef = new Map(population.map((p) => [p.ref, p]))
   142→   142→    return (savedSampling.refs ?? [])
   143→   143→      .map((ref) => byRef.get(ref))
   144→   144→      .filter((p): p is PopulationItem => Boolean(p))
   145→   145→      .map((p) => ({ ref: p.ref, book: p.book, status: 'PENDING' as const, error: 0 }))
   146→   146→  }, [savedSampling, sales.isSuccess, population])
   147→   147→
   148→   148→  const sampleItems = sampleOverride ?? savedSample
   149→   149→  const hvItems = hvOverride ?? savedSampling?.highValue ?? []
   150→   150→
   151→   151→  const seed = Math.abs(Number(seedStr.replace(/[^0-9]/g, '')) || 2024)
   152→   152→  const n = Math.max(1, Math.min(100, Number(nStr.replace(/[^0-9]/g, '')) || 25))
   153→   153→  const hvThreshold = Math.max(0, Number(hvStr.replace(/[^0-9]/g, '')) || 0)
   154→   154→
   155→   155→  // ---- Strates et population sondée ----
   156→   156→  const hvPop = useMemo(() => population.filter((p) => p.book >= hvThreshold && hvThreshold > 0), [population, hvThreshold])
   157→   157→  const samplingPop = useMemo(
   158→   158→    () => (hvThreshold > 0 ? population.filter((p) => p.book < hvThreshold) : population),
   159→   159→    [population, hvThreshold],
   160→   160→  )
   161→   161→  const samplingValue = samplingPop.reduce((s, p) => s + p.book, 0)
   162→   162→  const interval = samplingPop.length ? samplingValue / Math.max(1, sampleItems.length || n) : 0
   163→   163→
   164→   164→  const generate = () => {
   165→   165→    const sel = selectSample(method, samplingPop, n, seed)
   166→   166→    const byRef = new Map(samplingPop.map((p) => [p.ref, p]))
   167→   167→    setSampleOverride(sel.refs.map((ref) => toState(byRef.get(ref)!)).filter(Boolean))
   168→   168→    setHvOverride(hvPop.map((p) => ({ ref: p.ref, book: p.book, status: 'PENDING' as const, error: 0 })))
   169→   169→  }
   170→   170→
   171→   171→  const updateItem = (ref: string, patch: Partial<TestResult>, isHv = false) => {
   172→   172→    const setter = isHv ? setHvOverride : setSampleOverride
   173→   173→    const base = isHv ? hvItems : sampleItems
   174→   174→    setter(base.map((it) => (it.ref === ref ? { ...it, ...patch } : it)))
   175→   175→  }
   176→   176→
   177→   177→  // ---- Extrapolation ----
   178→   178→  const extrap = useMemo(
   179→   179→    () =>
   180→   180→      extrapolate({
   181→   181→        method,
   182→   182→        confidence,
   183→   183→        sample: sampleItems,
   184→   184→        populationValue: samplingValue,
   185→   185→        populationCount: samplingPop.length,
   186→   186→        highValueError: hvItems.filter((i) => i.status === 'ERROR').reduce((s, i) => s + i.error, 0),
   187→   187→      }),
   188→   188→    [method, confidence, sampleItems, samplingValue, samplingPop.length, hvItems],
   189→   189→  )
   190→   190→  const hvErrorCount = hvItems.filter((i) => i.status === 'ERROR').length
   191→   191→  const verdict = verdictAgainstThresholds(extrap.uel, extrap.projectedTotal, thresholds, extrap.errorCount + hvErrorCount)
   192→   192→  const testedCount = sampleItems.filter((i) => i.status !== 'PENDING').length + hvItems.filter((i) => i.status !== 'PENDING').length
   193→   193→  const totalCount = sampleItems.length + hvItems.length
   194→   194→  const suggestedN = useMemo(
   195→   195→    () => (samplingPop.length ? suggestedMusSize(samplingValue, thresholds.planification, confidence) : 0),
   196→   196→    [samplingPop.length, samplingValue, thresholds.planification, confidence],
   197→   197→  )
   198→   198→
   199→   199→  const saveSampling = () => {
   200→   200→    const payload: SamplingWorksheet = {
   201→   201→      population: samplingPop.length,
   202→   202→      taille: sampleItems.length,
   203→   203→      refs: sampleItems.map((i) => i.ref),
   204→   204→      method,
   205→   205→      seed,
   206→   206→      confidence,
   207→   207→      items: sampleItems,
   208→   208→      highValue: hvItems,
   209→   209→      computed: {
   210→   210→        interval,
   211→   211→        projected: extrap.projectedTotal,
   212→   212→        precision: extrap.precision,
   213→   213→        uel: extrap.uel,
   214→   214→        verdict: verdict.verdict,
   215→   215→      },
   216→   216→    }
   217→   217→    save.mutate({ section: 'sampling', data: payload })
   218→   218→  }
   219→   219→
   220→   220→  if (sales.isLoading) {
   221→   221→    return (
   222→   222→      <Card>
   223→   223→        <CardContent className="py-10 text-center text-sm text-muted-foreground">Chargement de la population des ventes…</CardContent>
   224→   224→      </Card>
   225→   225→    )
   226→   226→  }
   227→   227→
   228→   228→  const methodHint = METHODS.find((m) => m.value === method)?.hint
   229→   229→
   230→   230→  return (
   231→   231→    <Card className="print-avoid-break">
   232→   232→      <CardHeader>
   233→   233→        <CardTitle className="flex flex-wrap items-center gap-2 text-sm">
   234→   234→          <FlaskConical className="size-4 text-primary" />
   235→   235→          Échantillonnage & extrapolation statistique — cycle ventes (NEP 530)
   236→   236→          <Badge variant="outline" className="font-mono text-[10px]">
   237→   237→            {population.length} factures
   238→   238→          </Badge>
   239→   239→        </CardTitle>
   240→   240→      </CardHeader>
   241→   241→      <CardContent className="space-y-4">
   242→   242→        {/* ---------- Paramètres ---------- */}
   243→   243→        <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-5">
   244→   244→          <div className="space-y-1">
   245→   245→            <Label className="text-xs">Méthode de sélection</Label>
   246→   246→            <Select value={method} onValueChange={(v) => setMethod(v as SamplingMethod)}>
   247→   247→              <SelectTrigger aria-label="Méthode de sélection"><SelectValue /></SelectTrigger>
   248→   248→              <SelectContent>
   249→   249→                {METHODS.map((m) => (
   250→   250→                  <SelectItem key={m.value} value={m.value}>{m.label}</SelectItem>
   251→   251→                ))}
   252→   252→              </SelectContent>
   253→   253→            </Select>
   254→   254→          </div>
   255→   255→          <div className="space-y-1">
   256→   256→            <Label className="text-xs">Confiance</Label>
   257→   257→            <Select value={String(confidence)} onValueChange={(v) => setConfidence(Number(v) as ConfidenceLevel)}>
   258→   258→              <SelectTrigger aria-label="Niveau de confiance"><SelectValue /></SelectTrigger>
   259→   259→              <SelectContent>
   260→   260→                {CONFIDENCES.map((c) => (
   261→   261→                  <SelectItem key={c.value} value={c.value}>{c.label} · {c.z}</SelectItem>
   262→   262→                ))}
   263→   263→              </SelectContent>
   264→   264→            </Select>
   265→   265→          </div>
   266→   266→          <div className="space-y-1">
   267→   267→            <Label className="text-xs">Taille (n)</Label>
   268→   268→            <Input value={nStr} onChange={(e) => setNStr(e.target.value)} inputMode="numeric" aria-label="Taille d'échantillon" />
   269→   269→          </div>
   270→   270→          <div className="space-y-1">
   271→   271→            <Label className="text-xs">Graine (reproductible)</Label>
   272→   272→            <div className="flex gap-1">
   273→   273→              <Input value={seedStr} onChange={(e) => setSeedStr(e.target.value)} inputMode="numeric" aria-label="Graine du tirage" />
   274→   274→              <Button
   275→   275→                type="button"
   276→   276→                variant="outline"
   277→   277→                size="icon"
   278→   278→                aria-label="Graine aléatoire"
   279→   279→                title="Tirer une graine aléatoire"
   280→   280→                onClick={() => setSeedStr(String(1000 + Math.floor(Math.random() * 9000)))}
   281→   281→              >
   282→   282→                <Dices className="size-4" />
   283→   283→              </Button>
   284→   284→            </div>
   285→   285→          </div>
   286→   286→          <div className="space-y-1">
   287→   287→            <Label className="text-xs">Strate exhaustive (≥)</Label>
   288→   288→            <Input value={hvStr} onChange={(e) => setHvStr(e.target.value)} inputMode="numeric" aria-label="Seuil de stratification" />
   289→   289→          </div>
   290→   290→        </div>
   291→   291→        <p className="text-xs leading-relaxed text-muted-foreground">
   292→   292→          {methodHint} Les valeurs ≥ au seuil de stratification sont testées à 100 % (non extrapolées) ; les avoirs et
   293→   293→          opérations diverses (valeur ≤ 0) sont exclus de la population monétaire. Seuils de comparaison :{' '}
   294→   294→          {thresholds.saved ? 'feuille « Seuils » sauvegardée' : 'valeurs par défaut (feuille « Seuils » non sauvegardée)'}.
   295→   295→        </p>
   296→   296→        <div className="flex flex-wrap items-center gap-2">
   297→   297→          <Button size="sm" onClick={generate} disabled={sales.isLoading || samplingPop.length === 0} className="gap-1.5">
   298→   298→            <Crosshair className="size-3.5" /> Générer l’échantillon
   299→   299→          </Button>
   300→   300→          {samplingPop.length > 0 ? (
   301→   301→            <button
   302→   302→              type="button"
   303→   303→              className="inline-flex items-center gap-1.5 rounded-md border border-dashed px-2.5 py-1 text-xs text-muted-foreground transition-colors hover:border-primary/50 hover:text-foreground"
   304→   304→              onClick={() => setNStr(String(suggestedN))}
   305→   305→              title="Applique la taille d'échantillon MUS classique : n = V × P(0,γ) / seuil de planification"
   306→   306→            >
   307→   307→              Taille suggérée : <span className="font-semibold tabular-nums text-foreground">{suggestedN}</span> items
   308→   308→              <span className="text-[10px]">(appliquer)</span>
   309→   309→            </button>
   310→   310→          ) : null}
   311→   311→          {sampleItems.length > 0 ? (
   312→   312→            <Badge variant="secondary" className="tabular-nums">
   313→   313→              {testedCount}/{totalCount} items testés
   314→   314→            </Badge>
   315→   315→          ) : null}
   316→   316→        </div>
   317→   317→
   318→   318→        {/* ---------- Résumé de population ---------- */}
   319→   319→        {population.length > 0 ? (
   320→   320→          <div className="grid grid-cols-2 gap-2 rounded-lg border bg-muted/30 p-3 text-xs sm:grid-cols-5">
   321→   321→            <div>
   322→   322→              <p className="text-muted-foreground">Population</p>
   323→   323→              <p className="tabular-nums font-semibold">{samplingPop.length + hvPop.length} factures</p>
   324→   324→            </div>
   325→   325→            <div>
   326→   326→              <p className="text-muted-foreground">Valeur totale</p>
   327→   327→              <p className="tabular-nums font-semibold">{fmtEUR(samplingValue + hvPop.reduce((s, p) => s + p.book, 0))}</p>
   328→   328→            </div>
   329→   329→            <div>
   330→   330→              <p className="text-muted-foreground">Strate exhaustive</p>
   331→   331→              <p className="tabular-nums font-semibold">{hvPop.length} · {fmtEUR(hvPop.reduce((s, p) => s + p.book, 0))}</p>
   332→   332→            </div>
   333→   333→            <div>
   334→   334→              <p className="text-muted-foreground">Population sondée</p>
   335→   335→              <p className="tabular-nums font-semibold">{samplingPop.length} · {fmtEUR(samplingValue)}</p>
   336→   336→            </div>
   337→   337→            <div>
   338→   338→              <p className="text-muted-foreground">Intervalle J = V/n</p>
   339→   339→              <p className="tabular-nums font-semibold">{sampleItems.length ? fmtEUR(interval) : '—'}</p>
   340→   340→            </div>
   341→   341→          </div>
   342→   342→        ) : null}
   343→   343→
   344→   344→        {/* ---------- Strate exhaustive ---------- */}
   345→   345→        {hvItems.length > 0 ? (
   346→   346→          <div className="space-y-2">
   347→   347→            <h4 className="flex items-center gap-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   348→   348→              Strate des valeurs élevées — examen exhaustif
   349→   349→              <Badge variant="outline" className="normal-case">{hvItems.length} items</Badge>
   350→   350→            </h4>
   351→   351→            <ScrollArea className="print-expand max-h-56 rounded-md border">
   352→   352→              <table className="table-sticky w-full text-xs">
   353→   353→                <thead className="bg-muted">
   354→   354→                  <tr className="text-left uppercase text-muted-foreground">
   355→   355→                    <th className="p-2">Pièce</th>
   356→   356→                    <th className="p-2 text-right">Valeur</th>
   357→   357→                    <th className="p-2">Test</th>
   358→   358→                    <th className="p-2 text-right">Erreur</th>
   359→   359→                  </tr>
   360→   360→                </thead>
   361→   361→                <tbody>
   362→   362→                  {hvItems.map((it) => (
   363→   363→                    <ItemRow key={it.ref} it={it} onUpdate={(patch) => updateItem(it.ref, patch, true)} />
   364→   364→                  ))}
   365→   365→                </tbody>
   366→   366→              </table>
   367→   367→            </ScrollArea>
   368→   368→          </div>
   369→   369→        ) : null}
   370→   370→
   371→   371→        {/* ---------- Échantillon ---------- */}
   372→   372→        {sampleItems.length === 0 ? (
   373→   373→          <EmptyState
   374→   374→            icon={<Crosshair className="size-8" />}
   375→   375→            title="Aucun échantillon généré"
   376→   376→            description="Choisissez une méthode, une taille et une graine, puis générez : le tirage s’effectue réellement dans les factures de ventes du dossier. Saisissez ensuite vos résultats de test (conforme / anomalie + montant) pour obtenir l’extrapolation statistique."
   377→   377→          />
   378→   378→        ) : (
   379→   379→          <div className="space-y-2">
   380→   380→            <h4 className="flex items-center gap-2 text-xs font-semibold uppercase tracking-wide text-muted-foreground">
   381→   381→              Échantillon sondé
   382→   382→              <Badge variant="outline" className="normal-case">{sampleItems.length} items</Badge>
   383→   383→            </h4>
   384→   384→            <ScrollArea className="print-expand max-h-80 rounded-md border">
   385→   385→              <table className="table-sticky w-full text-xs">
   386→   386→                <thead className="bg-muted">
   387→   387→                  <tr className="text-left uppercase text-muted-foreground">
   388→   388→                    <th className="p-2">Pièce</th>
   389→   389→                    <th className="p-2 text-right">Valeur</th>
   390→   390→                    <th className="p-2">Test</th>
   391→   391→                    <th className="p-2 text-right">Erreur</th>
   392→   392→                    {method === 'mus' ? <th className="p-2 text-right">Tainting</th> : null}
   393→   393→                  </tr>
   394→   394→                </thead>
   395→   395→                <tbody>
   396→   396→                  {sampleItems.map((it) => (
   397→   397→                    <ItemRow key={it.ref} it={it} showTainting={method === 'mus'} onUpdate={(patch) => updateItem(it.ref, patch)} />
   398→   398→                  ))}
   399→   399→                </tbody>
   400→   400→              </table>
   401→   401→            </ScrollArea>
   402→   402→          </div>
   403→   403→        )}
   404→   404→
   405→   405→        {/* ---------- Extrapolation ---------- */}
   406→   406→        {sampleItems.length > 0 ? (
   407→   407→          <div className="space-y-3 rounded-lg border p-3">

   412→   412→              </Badge>
   413→   413→            </div>
   414→   414→            <div className="grid grid-cols-2 gap-2 text-xs sm:grid-cols-4">

   418→   418→                {extrap.highValueError > 0 ? (
   419→   419→                  <p className="text-[10px] text-muted-foreground">dont {fmtEUR(extrap.highValueError)} à 100 % (strate exhaustive)</p>
   420→   420→                ) : null}
   421→   421→              </div>
   422→   422→              <div className="rounded-md border p-2">
   423→   423→                <p className="text-muted-foreground">Précision statistique</p>

   434→   434→                </p>

   436→   436→                  <p className="text-[10px] text-muted-foreground">dont {hvErrorCount} en strate exhaustive</p>
   437→   437→                ) : null}
   438→   438→              </div>

   442→   442→            <ThresholdScale
   443→   443→              projected={extrap.projectedTotal}

   447→   447→
   448→   448→            <div

   451→   451→                  ? 'border-emerald-300 bg-emerald-50 text-emerald-900 dark:border-emerald-900 dark:bg-emerald-950 dark:text-emerald-200'

   457→   457→            >
   458→   458→              <p className="flex items-center gap-2 font-semibold">
   459→   459→                {verdict.verdict === 'ACCEPT' ? <ShieldCheck className="size-4" /> : verdict.verdict === 'ALERT' ? <TriangleAlert className="size-4" /> : <ShieldAlert className="size-4" />}

   461→   461→              </p>
   462→   462→              <p className="mt-1">{verdict.detail}</p>
   463→   463→              {method !== 'mus' ? (
   464→   464→                <p className="mt-1 flex gap-1.5 text-[11px] opacity-80">
   465→   465→                  <Info className="mt-0.5 size-3 shrink-0" />
   466→   466→                  Rappel méthodologique : l’estimation par la moyenne suppose une dispersion homogène des erreurs ; la
   467→   467→                  sélection systématique est souvent traitée avec les statistiques MPU — documentez ce choix.
   468→   468→                </p>
   469→   469→              ) : null}
   470→   470→            </div>
   471→   471→
   472→   472→            <Button size="sm" variant="outline" className="gap-1.5" onClick={saveSampling} disabled={save.isPending}>
   473→   473→              <Save className="size-3.5" /> Sauvegarder la feuille d’échantillonnage
   474→   474→            </Button>
   475→   475→            <Separator className="print:hidden" />
   476→   476→            <p className="text-[11px] leading-relaxed text-muted-foreground">
   477→   477→              Formules : MUS — J = V/n ; erreur projetée = Σ(tainting × J) ; UEL = J × [P(0) + Σ(P(k) − P(k−1)) × t₍ₖ₎]
   478→   478→              (borne de Stringer, facteurs de Poisson). MPU — projeté = moyenne des erreurs × N ; précision = z × s/√n × N.
   479→   479→              Les erreurs de la strate exhaustive s’ajoutent à 100 %. Graine {seed} : le tirage est reproductible.
   480→   480→            </p>
   481→   481→          </div>
   482→   482→        ) : null}
   483→   483→      </CardContent>
   484→   484→    </Card>
   485→   485→  )

   490→   490→function ItemRow({
   491→   491→  it,
   492→   492→  showTainting = false,
   493→   493→  onUpdate,
   494→   494→}: {
   495→   495→  it: TestResult
   496→   496→  showTainting?: boolean
   497→   497→  onUpdate: (patch: Partial<TestResult>) => void
   498→   498→}) {
   499→   499→  const tainting = it.book > 0 ? Math.min(1, Math.max(0, it.error) / it.book) : 0
   500→   500→  return (
   501→   501→    <tr className={`border-t ${it.status === 'ERROR' ? 'bg-red-50/60 dark:bg-red-950/30' : it.status === 'OK' ? '' : 'opacity-80'}`}>
   502→   502→      <td className="p-1.5 font-mono">{it.ref}</td>
   503→   503→      <td className="p-1.5 text-right tabular-nums">{fmtEUR(it.book)}</td>
   504→   504→      <td className="p-1.5">
   505→   505→        <Select value={it.status} onValueChange={(v) => onUpdate({ status: v as TestResult['status'], error: v === 'OK' ? 0 : it.error })}>
   506→   506→          <SelectTrigger className="h-7 w-32 text-xs" aria-label={`Résultat du test ${it.ref}`}>
   507→   507→            <SelectValue />
   508→   508→          </SelectTrigger>
   509→   509→          <SelectContent>
   510→   510→            <SelectItem value="PENDING">À tester</SelectItem>
   511→   511→            <SelectItem value="OK">Conforme</SelectItem>
   512→   512→            <SelectItem value="ERROR">Anomalie</SelectItem>
   513→   513→          </SelectContent>
   514→   514→        </Select>
   515→   515→      </td>
   516→   516→      <td className="p-1.5 text-right">
   517→   517→        {it.status === 'ERROR' ? (
   518→   518→          <Input
   519→   519→            className="h-7 w-28 text-right text-xs tabular-nums"
   520→   520→            value={String(it.error)}
   521→   521→            inputMode="decimal"
   522→   522→            aria-label={`Montant de l'erreur ${it.ref}`}
   523→   523→            onChange={(e) => {
   524→   524→              const v = Number(e.target.value.replace(',', '.').replace(/[^0-9.]/g, ''))
   525→   525→              onUpdate({ error: Number.isFinite(v) ? v : 0 })
   526→   526→            }}
   527→   527→          />
   528→   528→        ) : (
   529→   529→          <span className="text-muted-foreground">—</span>
   530→   530→        )}
   531→   531→      </td>
   532→   532→      {showTainting ? (
   533→   533→        <td className="p-1.5 text-right tabular-nums text-muted-foreground">
   534→   534→          {it.status === 'ERROR' ? `${(tainting * 100).toFixed(0)} %` : '—'}
   535→   535→        </td>
   536→   536→      ) : null}
   537→   537→    </tr>
   538→   538→  )
   539→   539→}

   543→   543→function ThresholdScale({
   544→   544→  projected,
   545→   545→  uel,
   546→   546→  thresholds,
   547→   547→}: {
   548→   548→  projected: number
   549→   549→  uel: number
   550→   550→  thresholds: { signification: number; planification: number; remontee: number }
   551→   551→}) {
   552→   552→  const max = Math.max(thresholds.signification * 1.15, uel * 1.05, 1)
   553→   553→  const pct = (v: number) => `${Math.min(100, (v / max) * 100)}%`
   554→   554→  const markers = [
   555→   555→    { v: thresholds.remontee, label: 'Remontée', cls: 'bg-muted-foreground/70' },
   556→   556→    { v: thresholds.planification, label: 'Planification', cls: 'bg-amber-500' },
   557→   557→    { v: thresholds.signification, label: 'Signification', cls: 'bg-red-500' },
   558→   558→  ]
   559→   559→  return (
   560→   560→    <div className="space-y-1.5" role="img" aria-label={`Erreur projetée ${fmtEUR(projected)}, limite supérieure ${fmtEUR(uel)}, seuil de signification ${fmtEUR(thresholds.signification)}`}>
   561→   561→      <div className="relative h-7 overflow-visible rounded-md border bg-muted/40">
   562→   562→        {/* Barre d'erreur projetée */}
   563→   563→        <div className="absolute inset-y-1 left-0 rounded-sm bg-amber-500/80 transition-all duration-500" style={{ width: pct(projected) }} />
   564→   564→        {/* Marqueur UEL */}
   565→   565→        <div
   566→   566→          className="absolute -top-0.5 bottom-0.5 w-0.5 bg-foreground transition-all duration-500"
   567→   567→          style={{ left: pct(uel) }}
   568→   568→          title={`UEL : ${fmtEUR(uel)}`}
   569→   569→        />
   570→   570→        {/* Marqueurs de seuils */}
   571→   571→        {markers.map((m) => (
   572→   572→          <div key={m.label} className={`absolute inset-y-0 w-0.5 ${m.cls}`} style={{ left: pct(m.v) }} title={`${m.label} : ${fmtEUR(m.v)}`} />
   573→   573→        ))}
   574→   574→      </div>
   575→   575→      <div className="flex flex-wrap items-center gap-x-4 gap-y-1 text-[10px] text-muted-foreground">
   576→   576→        <span className="flex items-center gap-1"><span className="inline-block h-2 w-3 rounded-sm bg-amber-500/80" /> Erreur projetée {fmtEUR(projected)}</span>
   577→   577→        <span className="flex items-center gap-1"><span className="inline-block h-3 w-0.5 bg-foreground" /> UEL {fmtEUR(uel)}</span>
   578→   578→        {markers.map((m) => (
   579→   579→          <span key={m.label} className="flex items-center gap-1">
   580→   580→            <span className={`inline-block h-2.5 w-0.5 ${m.cls}`} /> {m.label} {fmtEUR(m.v)}
   581→   581→          </span>
   582→   582→        ))}
   583→   583→      </div>
   584→   584→    </div>
   585→   585→  )
   586→   586→}
