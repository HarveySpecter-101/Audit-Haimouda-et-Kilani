   175→   175→/** Phrase de synthèse adaptée au signe de l'écart de tentatives entre

   177→   177→function comparisonTakeaway(current: number, prev: number): string {
   178→   178→  const diff = current - prev
   179→   179→  if (diff > 0) {
   180→   180→    return `Assiduité en hausse : +${diff} tentative${diff === 1 ? '' : 's'} par rapport aux 12 semaines précédentes.`

   182→   182→  if (diff < 0) {
   183→   183→    return `Assiduité en baisse : −${-diff} tentative${diff === -1 ? '' : 's'} par rapport aux 12 semaines précédentes.`
   184→   184→  }
   185→   185→  return current === 0
   186→   186→    ? 'Aucune tentative sur aucune des deux périodes — la série démarre avec vous.'
   187→   187→    : 'Assiduité stable : autant de tentatives que sur les 12 semaines précédentes.'
   188→   188→}

   190→   190→/** Couleur du delta (hausse / baisse / stable) — motif emerald/red établi
   191→   191→ *  dans tout le simulateur, muted pour l'égalité. */
   192→   192→function deltaTone(diff: number): string {
   193→   193→  if (diff > 0) return 'text-emerald-700 dark:text-emerald-400'
   194→   194→  if (diff < 0) return 'text-red-700 dark:text-red-400'
   195→   195→  return 'text-muted-foreground'
   196→   196→}

   200→   200→function ComparisonRow({ label, current, previous }: { label: string; current: number; previous: number }) {

   208→   208→        <span className="text-muted-foreground"> vs {previous}</span>

   219→   219→/** F-9c — constructeur pur du CSV (testable unitairement) : BOM UTF-8 pour
   220→   220→ *  Excel, séparateur point-virgule (Excel-FR), en-tête « Date;Tentatives »,
   221→   221→ *  une ligne par jour « YYYY-MM-DD;count », ligne de total en fin — fins de
   222→   222→ *  ligne CRLF (RFC 4180). Aucune dépendance, aucun accès DOM. */
   223→   223→export function buildActivityCsv(days: ActivityDay[]): string {
   224→   224→  const lines: string[] = ['Date;Tentatives']
   225→   225→  let total = 0
   226→   226→  for (const day of days) {
   227→   227→    lines.push(`${day.date};${day.count}`)
   228→   228→    total += day.count
   229→   229→  }
   230→   230→  lines.push(`Total;${total}`)
   231→   231→  return `\uFEFF${lines.join('\r\n')}\r\n`
   232→   232→}

   237→   237→function downloadActivityCsv(days: ActivityDay[]) {

   239→   239→  const url = URL.createObjectURL(blob)

   245→   245→  anchor.remove()
   246→   246→  URL.revokeObjectURL(url)
   247→   247→}

   256→   256→export function ActivityHeatmap({
   257→   257→  days,

   266→   266→}: ActivityHeatmapProps) {

   269→   269→  // sont intégralement DÉRIVÉES des props : zéro effet, zéro état dérivé

   274→   274→  // changer de journée retombe naturellement sur « Tous » sans le

   276→   276→  const [dayFilter, setDayFilter] = useState<{ date: string; mode: DayFilterMode }>({ date: '', mode: 'all' })
   277→   277→  // F-8d — horodatage du dernier « g » nu capté dans le composant :

   291→   291→      return res.json()
   292→   292→    },
   293→   293→    refetchOnWindowFocus: false,
   294→   294→    staleTime: 30_000,
   295→   295→  })
   296→   296→
   297→   297→  if (!days.length) return null
   298→   298→
   299→   299→  const todayN = todayCount ?? today ?? 0
   300→   300→  const best = bestDay ?? days.reduce((m, d) => Math.max(m, d.count), 0)
   301→   301→  // « aujourd'hui » = dernière entrée de days (convention du composant)
   302→   302→  const todayDate = days[days.length - 1].date
   303→   303→  const todayPanelOpen = selected === todayDate
   304→   304→
   305→   305→  // Analyse locale des dates — jamais new Date(iso) : pivot UTC selon le fuseau
   306→   306→  const cells = days.map((d) => {
   307→   307→    const [ys, ms, ds] = d.date.split('-')
   308→   308→    return { date: d.date, count: d.count, at: new Date(Number(ys), Number(ms) - 1, Number(ds)) }
   309→   309→  })
   310→   310→  // Alignement lundi : décalage de la première colonne (cellules vides avant la fenêtre)
   311→   311→  const offset = (cells[0].at.getDay() + 6) % 7
   312→   312→  const cols = Math.ceil((offset + cells.length) / 7)
   313→   313→
   314→   314→  const W = GX + cols * PITCH - GAP + 3
   315→   315→  const H = GY + 7 * PITCH - GAP + 3
   316→   316→
   317→   317→  // Libellé de mois au-dessus d'une colonne quand la semaine change de mois
   318→   318→  // (mois du DERNIER jour réel de la colonne — couvre un mois qui débute en
   319→   319→  // cours de semaine) ; garde-fou anti-chevauchement de libellés adjacents
   320→   320→  const monthLabels: { col: number; label: string }[] = []
   321→   321→  let lastLabelRight = -Infinity
   322→   322→  for (let c = 0; c < cols; c++) {
   323→   323→    const lastSlot = Math.min((c + 1) * 7 - 1, offset + cells.length - 1)
   324→   324→    const month = cells[lastSlot - offset].at.getMonth()
   325→   325→    const prevLastSlot = c === 0 ? -1 : Math.min(c * 7 - 1, offset + cells.length - 1)
   326→   326→    const prevMonth = prevLastSlot < offset ? -1 : cells[prevLastSlot - offset].at.getMonth()
   327→   327→    if (month === prevMonth) continue
   328→   328→    const label = MONTH_LABELS[month]
   329→   329→    const center = GX + c * PITCH + CELL / 2
   330→   330→    const halfW = (label.length * 4.6) / 2
   331→   331→    if (center - halfW < lastLabelRight + 1.5) continue
   332→   332→    monthLabels.push({ col: c, label })
   333→   333→    lastLabelRight = center + halfW
   334→   334→  }
   335→   335→
   336→   336→  const summary =
   337→   337→    `Assiduité sur ${cells.length} jours : ${total} tentatives au total, ${activeDays} jours actifs, ` +
   338→   338→    `${best} tentatives au maximum en une journée, série en cours de ${streak} jour${streak === 1 ? '' : 's'}, ` +
   339→   339→    `${todayN} tentative${todayN === 1 ? '' : 's'} aujourd'hui. Cliquez sur une journée active pour voir le détail des tentatives.`
   340→   340→
   341→   341→  const toggleDay = (date: string) => {
   342→   342→    setSelected((cur) => (cur === date ? null : date))
   343→   343→  }
   344→   344→
   345→   345→  // ---- F-8c : vue filtrée du panneau du jour (état dérivé, zéro effet) ----
   346→   346→  const filterMode: DayFilterMode = dayFilter.date === selected ? dayFilter.mode : 'all'
   347→   347→  const dayAttempts = dayQuery.data?.attempts ?? []
   348→   348→  const filteredAttempts =
   349→   349→    filterMode === 'all' ? dayAttempts : dayAttempts.filter((a) => a.mode === filterMode)
   350→   350→  const modeCounts: Record<Exclude<DayFilterMode, 'all'>, number> = { guided: 0, mission: 0, exam: 0 }
   351→   351→  for (const a of dayAttempts) {
   352→   352→    if (a.mode === 'guided' || a.mode === 'mission' || a.mode === 'exam') modeCounts[a.mode]++
   353→   353→  }
   354→   354→
   355→   355→  // ---- F-8d : raccourci local « D » (aujourd'hui) ----
   356→   356→  // POURQUOI un « d » nu est sûr : les raccourcis GLOBAUX
   357→   357→  // (keyboard-shortcuts.tsx) sont soit des SÉQUENCES « g » puis lettre
   358→   358→  // (≤ 900 ms), soit des touches isolées qui ne sont PAS « d » ([ ] / c ?
   359→   359→  // F1 — aucun Alt+lettre n'existe dans l'app). Un « d » tapé seul ne peut
   360→   360→  // donc déclencher aucune action globale. Seul cas d'intersection : « d »
   361→   361→  // comme SECONDE touche d'une séquence « g d » armée — on reflète localement
