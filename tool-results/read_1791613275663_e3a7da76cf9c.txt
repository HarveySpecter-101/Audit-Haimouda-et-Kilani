   109→function shuffle<T>(items: T[]): T[] {
   110→  const arr = [...items]
   111→  for (let i = arr.length - 1; i > 0; i--) {
   112→    const j = Math.floor(Math.random() * (i + 1))
   113→    ;[arr[i], arr[j]] = [arr[j], arr[i]]
   114→  }
   115→  return arr
   116→}

   118→/** File de session : cartes « à revoir » en premier, puis non vues, puis maîtrisées. */
   119→function buildQueue(cards: Flashcard[], progress: FlashcardsProgress): Flashcard[] {
   120→  const toReview = shuffle(cards.filter((c) => progress.review[c.id]))
   121→  const unseen = shuffle(cards.filter((c) => !progress.review[c.id] && !progress.known[c.id]))
   122→  const known = shuffle(cards.filter((c) => progress.known[c.id]))
   123→  return [...toReview, ...unseen, ...known]
   124→}
